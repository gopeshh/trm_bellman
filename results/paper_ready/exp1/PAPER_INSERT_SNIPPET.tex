% Figure: Unroll Sensitivity (B0 initial states)
\begin{figure}[t]
    \centering
    \includegraphics[width=\linewidth]{figures/fig_exp1_unroll_sensitivity_main.pdf}
    \caption{Unroll sensitivity on B0 (initial states). Contraction reduces
    value drift ($\Delta_V$) by 4.1$\times$, policy divergence ($\Delta_\pi$)
    by 33$\times$, and latent drift ($\Delta_z$) by 3.1$\times$ at 8$\times$
    training depth.}
    \label{fig:unroll_sensitivity}
\end{figure}

% Figure: Radius Sweep (B0 initial states)
\begin{figure}[t]
    \centering
    \includegraphics[width=\linewidth]{figures/fig_exp1_radius_sweep_main.pdf}
    \caption{Radius sweep on B0 (initial states). Even with projection disabled
    (R=0), contraction provides 4.8$\times$ value stability improvement
    ($\Delta_V$: 1.084 $\to$ 0.225), demonstrating stability comes from
    contraction, not projection clipping.}
    \label{fig:radius_sweep}
\end{figure}
